# Build the pinned MuseScore modules, without configuring the Studio application.
set(JIANPU_MUSESCORE_SOURCE "${CMAKE_SOURCE_DIR}/references/MuseScore" CACHE PATH "Pinned MuseScore source")
set(_revision 4509c9072c583f7ab0c805ded2ab88f29e0ec0e5)
find_package(Git REQUIRED)
if(EXISTS "${JIANPU_MUSESCORE_SOURCE}/.git")
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${JIANPU_MUSESCORE_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE _actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT _actual STREQUAL _revision)
    message(FATAL_ERROR "MuseScore revision differs: expected ${_revision}, got ${_actual}")
endif()
foreach(_entry "muse|42b02245d36f1f37dd2b04a382e27c28c9ece006" "muse_deps|a9db17d2f0478859711873f2db156eedfc082040")
    string(REPLACE "|" ";" _pair "${_entry}")
    list(GET _pair 0 _dir)
    list(GET _pair 1 _sha)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${JIANPU_MUSESCORE_SOURCE}/${_dir}" rev-parse HEAD
        OUTPUT_VARIABLE _actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT _actual STREQUAL _sha)
        message(FATAL_ERROR "${_dir} revision differs; run scripts/fetch-reference.sh")
    endif()
endforeach()
else()
    # Corresponding-source tarballs carry the same revision manifest without .git.
    file(STRINGS "${JIANPU_MUSESCORE_SOURCE}/JIANPU_SOURCE_LOCK" _archive_lock)
    set(_expected_lock "4509c9072c583f7ab0c805ded2ab88f29e0ec0e5;42b02245d36f1f37dd2b04a382e27c28c9ece006;a9db17d2f0478859711873f2db156eedfc082040")
    if(NOT _archive_lock STREQUAL _expected_lock)
        message(FATAL_ERROR "Missing or incompatible source archive revision manifest")
    endif()
endif()

# The pinned 5.00 reader probes every note property as a potential spanner.
# A failed probe is normal (e.g. an eid); the subsequent property reader still
# reports genuinely unknown tags. Make that probe silent, without hiding errors.
file(READ "${JIANPU_MUSESCORE_SOURCE}/src/engraving/rw/read500/tread.cpp" _reader)
string(REPLACE "TConv::fromXml(tag, ElementType::INVALID);"
    "TConv::fromXml(tag, ElementType::INVALID, true);" _reader "${_reader}")
string(REPLACE "#include \"../../" "#include \"engraving/" _reader "${_reader}")
string(REPLACE "#include \"../" "#include \"engraving/rw/" _reader "${_reader}")
string(REPLACE "#include \"tread.h\"" "#include \"engraving/rw/read500/tread.h\"" _reader "${_reader}")
string(REPLACE "#include \"readcontext.h\"" "#include \"engraving/rw/read500/readcontext.h\"" _reader "${_reader}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/tread500.cpp" CONTENT "${_reader}")
target_sources(engraving PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/tread500.cpp")
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/tread500.cpp" PROPERTIES SKIP_UNITY_BUILD_INCLUSION ON)
add_subdirectory("${JIANPU_MUSESCORE_SOURCE}/src/importexport/musicxml" musicxml)
